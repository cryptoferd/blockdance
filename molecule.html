Desktop: drag to look, scroll to zoom.
Quest 2: open in Oculus Browser → press Enter VR.
VR: point at molecule + hold trigger to grab/release.
import { VRButton } from "https://unpkg.com/three@0.160.0/examples/jsm/webxr/VRButton.js"; import { OrbitControls } from "https://unpkg.com/three@0.160.0/examples/jsm/controls/OrbitControls.js"; import { XRControllerModelFactory } from "https://unpkg.com/three@0.160.0/examples/jsm/webxr/XRControllerModelFactory.js"; // --------------------------- // Three.js + WebXR setup // --------------------------- const scene = new THREE.Scene(); scene.fog = new THREE.Fog(0x000000, 2, 18); const camera = new THREE.PerspectiveCamera(65, window.innerWidth / window.innerHeight, 0.05, 200); camera.position.set(0, 1.45, 2.25); const renderer = new THREE.WebGLRenderer({ antialias: true }); renderer.setSize(window.innerWidth, window.innerHeight); renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2)); renderer.xr.enabled = true; document.body.appendChild(renderer.domElement); // VR Button (Quest 2 will show Enter VR) document.body.appendChild(VRButton.createButton(renderer)); // Desktop orbit controls (ignored in VR) const controls = new OrbitControls(camera, renderer.domElement); controls.target.set(0, 1.35, 0); controls.enableDamping = true; controls.dampingFactor = 0.06; controls.update(); // Lighting scene.add(new THREE.HemisphereLight(0x88aaff, 0x110011, 0.9)); const key = new THREE.DirectionalLight(0xffffff, 1.25); key.position.set(2, 4, 2); scene.add(key); // Floor + grid reference const floor = new THREE.Mesh( new THREE.PlaneGeometry(50, 50), new THREE.MeshStandardMaterial({ color: 0x060606, roughness: 1 }) ); floor.rotation.x = -Math.PI / 2; floor.position.y = 0; scene.add(floor); const grid = new THREE.GridHelper(20, 40, 0x222222, 0x111111); grid.position.y = 0.001; scene.add(grid); // Starfield for immersion { const starCount = 1500; const pos = new Float32Array(starCount * 3); for (let i = 0; i < starCount; i++) { const r = 8 + Math.random() * 40; const theta = Math.random() * Math.PI * 2; const phi = Math.acos(2 * Math.random() - 1); pos[i*3+0] = r * Math.sin(phi) * Math.cos(theta); pos[i*3+1] = (r * Math.cos(phi)) * 0.6 + 1.0; pos[i*3+2] = r * Math.sin(phi) * Math.sin(theta); } const g = new THREE.BufferGeometry(); g.setAttribute("position", new THREE.BufferAttribute(pos, 3)); const m = new THREE.PointsMaterial({ size: 0.02 }); scene.add(new THREE.Points(g, m)); } // --------------------------- // H2O molecule // --------------------------- const molecule = new THREE.Group(); molecule.position.set(0, 1.35, 0); scene.add(molecule); const bondLength = 0.55; // visual units (VR meters-ish) const bondAngleDeg = 104.5; // H-O-H const halfAngle = THREE.MathUtils.degToRad(bondAngleDeg / 2); const matO = new THREE.MeshStandardMaterial({ color: 0xff2b2b, roughness: 0.35, metalness: 0.05 }); const matH = new THREE.MeshStandardMaterial({ color: 0xf4f4f4, roughness: 0.55, metalness: 0.02 }); const matBond = new THREE.MeshStandardMaterial({ color: 0x8aa1ff, roughness: 0.6, metalness: 0.05 }); const oxygen = new THREE.Mesh(new THREE.SphereGeometry(0.22, 40, 24), matO); oxygen.material.emissive = new THREE.Color(0x330000); molecule.add(oxygen); const h1Pos = new THREE.Vector3( Math.sin(halfAngle) * bondLength, 0, Math.cos(halfAngle) * bondLength ); const h2Pos = new THREE.Vector3( -Math.sin(halfAngle) * bondLength, 0, Math.cos(halfAngle) * bondLength ); const hydrogen1 = new THREE.Mesh(new THREE.SphereGeometry(0.14, 32, 18), matH); hydrogen1.position.copy(h1Pos); hydrogen1.material.emissive = new THREE.Color(0x101010); const hydrogen2 = new THREE.Mesh(new THREE.SphereGeometry(0.14, 32, 18), matH); hydrogen2.position.copy(h2Pos); hydrogen2.material.emissive = new THREE.Color(0x101010); molecule.add(hydrogen1, hydrogen2); function makeBond(a, b) { const dir = new THREE.Vector3().subVectors(b, a); const len = dir.length(); const geo = new THREE.CylinderGeometry(0.04, 0.04, len, 18, 1, true); const bond = new THREE.Mesh(geo, matBond); const mid = new THREE.Vector3().addVectors(a, b).multiplyScalar(0.5); bond.position.copy(mid); bond.quaternion.setFromUnitVectors( new THREE.Vector3(0, 1, 0), dir.clone().normalize() ); return bond; } molecule.add(makeBond(new THREE.Vector3(0,0,0), h1Pos)); molecule.add(makeBond(new THREE.Vector3(0,0,0), h2Pos)); // --------------------------- // p5 HUD -> CanvasTexture -> floating plane // --------------------------- const hudW = 512, hudH = 256; let hudG = null; // create a dummy texture now; we swap its image once p5 is ready const hudTex = new THREE.CanvasTexture(document.createElement("canvas")); hudTex.minFilter = THREE.LinearFilter; hudTex.magFilter = THREE.LinearFilter; const hudPlane = new THREE.Mesh( new THREE.PlaneGeometry(0.9, 0.45), new THREE.MeshBasicMaterial({ map: hudTex, transparent: true }) ); hudPlane.position.set(0, 1.7, -1.2); scene.add(hudPlane); // p5 instance mode new window.p5((p) => { p.setup = () => { hudG = p.createGraphics(hudW, hudH); const c = p.createCanvas(1, 1); c.elt.style.display = "none"; }; p.draw = () => {}; }); function updateHud(timeSec) { if (!hudG) return; hudG.clear(); hudG.noStroke(); hudG.fill(0, 0, 0, 170); hudG.rect(0, 0, hudW, hudH, 18); hudG.fill(255); hudG.textAlign(hudG.CENTER, hudG.TOP); hudG.textSize(34); hudG.text("H₂O — Water Molecule", hudW/2, 16); hudG.textAlign(hudG.LEFT, hudG.TOP); hudG.textSize(20); hudG.fill(220); hudG.text(`Bond angle: ${bondAngleDeg.toFixed(1)}°`, 22, 78); hudG.text(`Bond length (visual): ${bondLength.toFixed(2)} units`, 22, 108); hudG.fill(180); hudG.textSize(18); hudG.text("VR: Trigger = grab • Move/rotate with controller", 22, 158); const pulse = 0.5 + 0.5 * Math.sin(timeSec * 2.3); hudG.fill(120, 180, 255, 200); hudG.rect(22, 205, (hudW - 44) * pulse, 18, 8); hudTex.image = hudG.elt; hudTex.needsUpdate = true; } function billboardHUD() { const camPos = new THREE.Vector3(); camera.getWorldPosition(camPos); hudPlane.lookAt(camPos); } // --------------------------- // Controllers + grabbing // --------------------------- const controllerModelFactory = new XRControllerModelFactory(); const raycaster = new THREE.Raycaster(); const tempMatrix = new THREE.Matrix4(); let grabbedBy = null; function buildController(i) { const controller = renderer.xr.getController(i); // visible ray const lineGeo = new THREE.BufferGeometry().setFromPoints([ new THREE.Vector3(0, 0, 0), new THREE.Vector3(0, 0, -1) ]); const line = new THREE.Line(lineGeo, new THREE.LineBasicMaterial()); line.scale.z = 2.0; controller.add(line); controller.addEventListener("selectstart", () => { tempMatrix.identity().extractRotation(controller.matrixWorld); raycaster.ray.origin.setFromMatrixPosition(controller.matrixWorld); raycaster.ray.direction.set(0, 0, -1).applyMatrix4(tempMatrix); const hits = raycaster.intersectObject(molecule, true); if (hits.length) { grabbedBy = controller; // Preserve world transform while re-parenting molecule.updateMatrixWorld(true); const worldPos = new THREE.Vector3(); const worldQuat = new THREE.Quaternion(); const worldScale = new THREE.Vector3(); molecule.matrixWorld.decompose(worldPos, worldQuat, worldScale); controller.add(molecule); molecule.position.copy(controller.worldToLocal(worldPos.clone())); molecule.quaternion.copy(worldQuat); molecule.scale.copy(worldScale); } }); controller.addEventListener("selectend", () => { if (grabbedBy === controller) { molecule.updateMatrixWorld(true); const worldPos = new THREE.Vector3(); const worldQuat = new THREE.Quaternion(); const worldScale = new THREE.Vector3(); molecule.matrixWorld.decompose(worldPos, worldQuat, worldScale); scene.add(molecule); molecule.position.copy(worldPos); molecule.quaternion.copy(worldQuat); molecule.scale.copy(worldScale); grabbedBy = null; } }); scene.add(controller); const grip = renderer.xr.getControllerGrip(i); grip.add(controllerModelFactory.createControllerModel(grip)); scene.add(grip); } buildController(0); buildController(1); // --------------------------- // Animate // --------------------------- const baseRotY = molecule.rotation.y; renderer.setAnimationLoop((t) => { const timeSec = t * 0.001; if (!grabbedBy) { molecule.rotation.y = baseRotY + Math.sin(timeSec * 0.6) * 0.25; oxygen.scale.setScalar(1.0 + Math.sin(timeSec * 2.2) * 0.015); hydrogen1.position.copy(h1Pos).add(new THREE.Vector3(0, Math.sin(timeSec * 3.0) * 0.01, 0)); hydrogen2.position.copy(h2Pos).add(new THREE.Vector3(0, Math.sin(timeSec * 3.2 + 1.3) * 0.01, 0)); } updateHud(timeSec); billboardHUD(); controls.update(); renderer.render(scene, camera); }); // Resize window.addEventListener("resize", () => { camera.aspect = window.innerWidth / window.innerHeight; camera.updateProjectionMatrix(); renderer.setSize(window.innerWidth, window.innerHeight); });
